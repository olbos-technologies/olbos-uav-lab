cmake_minimum_required(VERSION 3.22)
project(olbos_uav_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(uav_core INTERFACE)
target_include_directories(uav_core INTERFACE core/include)
add_executable(core_test tests/unit/core_test.cpp)
target_link_libraries(core_test PRIVATE uav_core)
target_compile_options(core_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
add_test(NAME core_contracts COMMAND core_test)
add_executable(uav_demo core/demo.cpp)
target_link_libraries(uav_demo PRIVATE uav_core)
add_executable(uav_scenarios core/scenario_runner.cpp)
target_link_libraries(uav_scenarios PRIVATE uav_core)
target_compile_options(uav_scenarios PRIVATE -Wall -Wextra -Wpedantic -Werror)

option(UAV_BUILD_GTEST "Build GoogleTest mission controller tests" OFF)
if(UAV_BUILD_GTEST)
  find_package(GTest REQUIRED)
  add_executable(mission_controller_test tests/unit/mission_controller_test.cpp)
  target_link_libraries(mission_controller_test PRIVATE uav_core GTest::gtest_main)
  target_compile_options(mission_controller_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_executable(scenario_test tests/unit/scenario_test.cpp)
  target_link_libraries(scenario_test PRIVATE uav_core GTest::gtest_main)
  target_compile_options(scenario_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_compile_definitions(scenario_test PRIVATE UAV_SOURCE_DIR="${CMAKE_SOURCE_DIR}")
  include(GoogleTest)
  gtest_discover_tests(mission_controller_test)
  gtest_discover_tests(scenario_test)
endif()
