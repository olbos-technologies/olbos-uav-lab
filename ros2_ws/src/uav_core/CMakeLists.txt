cmake_minimum_required(VERSION 3.22)
project(uav_core LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
add_library(uav_core INTERFACE)
target_compile_features(uav_core INTERFACE cxx_std_20)
target_include_directories(uav_core INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/../../../core/include> $<INSTALL_INTERFACE:include>)
install(DIRECTORY ../../../core/include/ DESTINATION include)
install(TARGETS uav_core EXPORT uav_coreTargets)
ament_export_targets(uav_coreTargets HAS_LIBRARY_TARGET)
ament_package()
