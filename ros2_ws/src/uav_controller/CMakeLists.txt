cmake_minimum_required(VERSION 3.22)
project(uav_controller)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(px4_msgs REQUIRED)
add_executable(sitl_mission src/sitl_mission.cpp)
target_compile_features(sitl_mission PRIVATE cxx_std_20)
target_include_directories(sitl_mission PRIVATE ../../../core/include)
ament_target_dependencies(sitl_mission rclcpp px4_msgs)
target_compile_options(sitl_mission PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS sitl_mission DESTINATION lib/${PROJECT_NAME})
ament_package()
