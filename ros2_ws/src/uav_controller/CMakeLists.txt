cmake_minimum_required(VERSION 3.22)
project(uav_controller)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_srvs REQUIRED)
find_package(uav_core REQUIRED)
find_package(px4_msgs REQUIRED)
add_executable(sitl_mission src/sitl_mission.cpp)
target_compile_features(sitl_mission PRIVATE cxx_std_20)
target_link_libraries(sitl_mission uav_core::uav_core)
ament_target_dependencies(sitl_mission rclcpp px4_msgs std_srvs)
target_compile_options(sitl_mission PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS sitl_mission DESTINATION lib/${PROJECT_NAME})
ament_package()
